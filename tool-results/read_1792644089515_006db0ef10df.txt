   118→/** api() declares `body` via `RequestInit & { body?: unknown }` — the intersection collapses to
   119→ * BodyInit, so JSON payloads go through this adapter (api() JSON-encodes them itself). */
   120→function postJson<T>(path: string, payload: unknown, headers: Record<string, string> = {}): Promise<T> {
   121→  return api<T>(path, { method: "POST", body: payload as BodyInit, headers });

   159→/* ── shared role hook (server-enforced; UI only hides what it must) ─────────── */

   161→function useRole(): string | null {
   162→  const { actor } = useAppStore();
   163→  const { data } = useQuery<SystemState>({
   164→    queryKey: ["system-state", actor],
   165→    queryFn: () => api<SystemState>(v1("/system/state")),
   166→    staleTime: 30_000,
   167→  });
   168→  return data?.actors.find((a) => a.id === actor)?.role ?? null;

   173→function FlagsTab() {
   174→  const { locale, iso } = useI18n();
   175→  const queryClient = useQueryClient();
   176→  const role = useRole();
   177→  const privileged = role === "admin" || role === "compliance_lead";
   178→  const [editing, setEditing] = useState<FlagItem | null>(null);
   179→  const [editValue, setEditValue] = useState("");
   180→  const [reason, setReason] = useState("");
   181→  const [error, setError] = useState<string | null>(null);
   182→
   183→  const query = useQuery<{ items: FlagItem[] }>({
   184→    queryKey: ["flags"],
   185→    queryFn: () => api<{ items: FlagItem[] }>(v1("/flags")),
   186→  });
   187→
   188→  const save = useMutation({
   189→    mutationFn: (body: { key: string; value: string; reason: string }) =>
   190→      postJson<{ ok: boolean }>(v1("/flags"), body, { "idempotency-key": crypto.randomUUID() }),
   191→    onSuccess: () => {
   192→      setEditing(null);
   193→      setError(null);
   194→      queryClient.invalidateQueries({ queryKey: ["flags"] });
   195→      queryClient.invalidateQueries({ queryKey: ["system-state"] });
   196→    },
   197→    onError: (e) => setError(e instanceof ApiError ? e.message : (e as Error).message),
   198→  });
   199→
   200→  const openEdit = (f: FlagItem): void => {
   201→    setEditing(f);
   202→    setEditValue(f.value);
   203→    setReason("");
   204→    setError(null);
   205→  };
   206→
   207→  const onSave = (): void => {
   208→    if (!editing) return;
   209→    if (reason.trim().length < 5) {
   210→      setError("Every flag change needs a reason (at least 5 characters).");
   211→      return;
   212→    }
   213→    save.mutate({ key: editing.key, value: editValue, reason: reason.trim() });
   214→  };
   215→
   216→  const isBoolean = (v: string): boolean => v === "true" || v === "false";
   217→
   218→  return (
   219→    <Card>
   220→      <CardHeader className="pb-3">
   221→        <CardTitle className="text-sm">Feature flags</CardTitle>
   222→        <CardDescription>Every change is audited with who changed what, when, and why. Effective within 10 seconds.</CardDescription>
   223→      </CardHeader>
   224→      <CardContent className="space-y-3">
   225→        {!privileged ? (
   226→          <Alert className="border-hold/40 bg-hold/10 text-foreground" role="note">
   227→            <TriangleAlert className="h-4 w-4 text-hold" aria-hidden />
   228→            <AlertTitle>Read-only</AlertTitle>
   229→            <AlertDescription>Only an administrator or the compliance lead can change flags.</AlertDescription>
   230→          </Alert>
   231→        ) : null}
   232→
   233→        {query.isLoading ? <LoadingSkeleton rows={6} /> : null}
   234→        {query.isError ? (
   235→          <ErrorState message={(query.error as Error).message} code={query.error instanceof ApiError ? query.error.code : undefined} onRetry={() => query.refetch()} />
   236→        ) : null}
   237→        {query.data && query.data.items.length === 0 ? <EmptyState title="No flags defined" icon={<Flag className="h-8 w-8" aria-hidden />} /> : null}
   238→
   239→        {query.data && query.data.items.length > 0 ? (
   240→          <div className="max-h-[60vh] overflow-y-auto rounded-lg border">
   241→            <Table>
   242→              <TableHeader className="sticky top-0 z-10 bg-card shadow-[0_1px_0_0_var(--border)]">
   243→                <TableRow>
   244→                  <TableHead>Key</TableHead>
   245→                  <TableHead className="w-28">Value</TableHead>
   246→                  <TableHead className="w-36">Updated by</TableHead>
   247→                  <TableHead>Reason</TableHead>
   248→                  <TableHead className="w-36">Updated</TableHead>
   249→                  {privileged ? <TableHead className="w-16 text-end">Edit</TableHead> : null}
   250→                </TableRow>
   251→              </TableHeader>
   252→              <TableBody>
   253→                {query.data.items.map((f) => (
   254→                  <TableRow key={f.key}>
   255→                    <TableCell className="font-mono text-xs">{f.key}</TableCell>
   256→                    <TableCell>
   257→                      <Badge variant="outline" className={cn("max-w-28 border-border bg-secondary font-mono text-[11px]", f.value === "true" && "border-ok/40 bg-ok/10", f.value === "false" && "text-muted-foreground")}>
   258→                        <span className="truncate">{trunc(f.value, 20)}</span>
   259→                      </Badge>
   260→                    </TableCell>
   261→                    <TableCell className="text-xs">{f.updatedBy}</TableCell>
   262→                    <TableCell className="text-xs text-muted-foreground">{f.reason}</TableCell>
   263→                    <TableCell className="whitespace-nowrap text-xs">{fmtTime(f.updatedAt, locale)}</TableCell>
   264→                    {privileged ? (
   265→                      <TableCell className="text-end">
   266→                        <Button size="sm" variant="ghost" onClick={() => openEdit(f)} aria-label={`Edit flag ${f.key}`}>
   267→                          <Settings2 className="h-3.5 w-3.5" aria-hidden />
   268→                        </Button>
   269→                      </TableCell>
   270→                    ) : null}
   271→                  </TableRow>
   272→                ))}
   273→              </TableBody>
   274→            </Table>
   275→          </div>
   276→        ) : null}
   277→        <p className="text-[11px] text-muted-foreground">{iso(String(query.data?.items.length ?? 0))} flags</p>
   278→      </CardContent>
   279→
   280→      <Dialog open={editing !== null} onOpenChange={(open) => !open && setEditing(null)}>
   281→        <DialogContent className="sm:max-w-md">
   282→          <DialogHeader>
   283→            <DialogTitle className="font-mono text-sm">{editing?.key}</DialogTitle>
   284→            <DialogDescription>Changing a flag takes effect within 10 seconds and is written to the audit trail with your reason.</DialogDescription>
   285→          </DialogHeader>
   286→          <div className="space-y-3">
   287→            {editing && isBoolean(editing.value) ? (
   288→              <div className="space-y-1.5">
   289→                <Label htmlFor="flag-value" className="text-xs">New value</Label>
   290→                <Select value={editValue} onValueChange={setEditValue}>
   291→                  <SelectTrigger id="flag-value" size="sm"><SelectValue /></SelectTrigger>
   292→                  <SelectContent>
   293→                    <SelectItem value="true">true</SelectItem>
   294→                    <SelectItem value="false">false</SelectItem>
   295→                  </SelectContent>
   296→                </Select>
   297→              </div>
   298→            ) : (
   299→              <div className="space-y-1.5">
   300→                <Label htmlFor="flag-value" className="text-xs">New value</Label>
   301→                <Input id="flag-value" value={editValue} onChange={(e) => setEditValue(e.target.value)} className="font-mono text-xs" aria-label="New flag value" />
   302→              </div>
   303→            )}
   304→            <div className="space-y-1.5">
   305→              <Label htmlFor="flag-reason" className="text-xs">Reason (required, ≥ 5 characters)</Label>
   306→              <Textarea id="flag-reason" value={reason} onChange={(e) => setReason(e.target.value)} rows={2} aria-label="Reason for the flag change" />
   307→            </div>
   308→            {error ? (
   309→              <p className="text-xs text-destructive" role="alert">{error}</p>
   310→            ) : null}
   311→          </div>
   312→          <DialogFooter>
   313→            <Button variant="outline" size="sm" onClick={() => setEditing(null)}>Cancel</Button>
   314→            <Button size="sm" onClick={onSave} disabled={save.isPending || reason.trim().length < 5}>
   315→              {save.isPending ? <Loader2 className="me-1.5 h-3.5 w-3.5 animate-spin" aria-hidden /> : null}
   316→              Save flag
   317→            </Button>
   318→          </DialogFooter>
   319→        </DialogContent>
   320→      </Dialog>
   321→    </Card>
   322→  );

   327→function ModesTab() {
   328→  const { t, locale, iso } = useI18n();
   329→  const queryClient = useQueryClient();
   330→  const role = useRole();
   331→  const privileged = role === "admin" || role === "compliance_lead";
   332→  const [nextMode, setNextMode] = useState<string>("M0");
   333→  const [reason, setReason] = useState("");
   334→  const [error, setError] = useState<string | null>(null);
   335→  const [note, setNote] = useState<string | null>(null);
   336→
   337→  const query = useQuery<ModeState>({
   338→    queryKey: ["system-mode"],
   339→    queryFn: () => api<ModeState>(v1("/system/mode")),
   340→  });
   341→
   342→  const setMode = useMutation({
   343→    mutationFn: (body: { mode: string; reason: string }) =>
   344→      postJson<ModeState>(v1("/system/mode"), body, { "idempotency-key": crypto.randomUUID() }),
   345→    onSuccess: (data) => {
   346→      setError(null);
   347→      setNote(`Mode set to ${data.current} — ${data.label}.`);
   348→      setReason("");
   349→      queryClient.invalidateQueries({ queryKey: ["system-mode"] });
   350→      queryClient.invalidateQueries({ queryKey: ["system-state"] });
   351→    },
   352→    onError: (e) => {
   353→      setNote(null);
   354→      setError(e instanceof ApiError ? e.message : (e as Error).message);
   355→    },
   356→  });
   357→
   358→  const onSet = (): void => {
   359→    if (reason.trim().length < 5) {
   360→      setError("A reason of at least 5 characters is required.");
   361→      return;
   362→    }
   363→    setMode.mutate({ mode: nextMode, reason: reason.trim() });
   364→  };
   365→
   366→  const current = query.data?.current ?? "M0";
   367→
   368→  return (
   369→    <div className="grid gap-4 lg:grid-cols-2">
   370→      <Card>
   371→        <CardHeader className="pb-3">
   372→          <CardTitle className="text-sm">Failsafe modes</CardTitle>

   375→        <CardContent className="space-y-4">
   376→          <div className="flex flex-wrap items-center gap-2">
   377→            <ModeChip mode={current} />
   378→            <span className="text-xs text-muted-foreground">
   379→              since {fmtTime(query.data?.since ?? null, locale)}
   380→              {query.data?.reason ? ` · reason: ${query.data.reason}` : ""}
   381→            </span>
   382→          </div>
   383→          <Separator />
   384→          <ul className="space-y-2.5">
   385→            {MODES.map((m) => (
   386→              <li key={m} className={cn("flex gap-2 rounded-lg border p-2.5", m === current ? "border-primary/40 bg-secondary" : "border-border")}>
   387→                <ModeChip mode={m} />
   388→                <p className="flex-1 text-xs text-muted-foreground">{t(`mode.${m}`)} — {locale === "ar" ? MODE_INFO[m]!.ar : MODE_INFO[m]!.en}</p>
   389→              </li>
   390→            ))}
   391→          </ul>
   392→        </CardContent>
   393→      </Card>
   394→
   395→      <Card>
   396→        <CardHeader className="pb-3">
   397→          <CardTitle className="text-sm">Operator switcher</CardTitle>
   398→          <CardDescription>Admin and compliance lead only — enforced server-side; every switch is audited with a reason.</CardDescription>
   399→        </CardHeader>
   400→        <CardContent className="space-y-3">
   401→          {privileged ? (
   402→            <>
   403→              <div className="space-y-1.5">
   404→                <Label htmlFor="mode-select" className="text-xs">Mode</Label>
   405→                <Select value={nextMode} onValueChange={setNextMode}>
   406→                  <SelectTrigger id="mode-select" size="sm"><SelectValue /></SelectTrigger>
   407→                  <SelectContent>
   408→                    {MODES.map((m) => (
   409→                      <SelectItem key={m} value={m}>{m} · {t(`mode.${m}`)}</SelectItem>
   410→                    ))}
   411→                  </SelectContent>
   412→                </Select>
   413→              </div>
   414→              <div className="space-y-1.5">
   415→                <Label htmlFor="mode-reason" className="text-xs">Reason (required, ≥ 5 characters)</Label>
   416→                <Input id="mode-reason" value={reason} onChange={(e) => setReason(e.target.value)} aria-label="Reason for the mode change" />
   417→              </div>
   418→              <Button size="sm" onClick={onSet} disabled={setMode.isPending || reason.trim().length < 5}>
   419→                {setMode.isPending ? <Loader2 className="me-1.5 h-3.5 w-3.5 animate-spin" aria-hidden /> : <Settings2 className="me-1.5 h-3.5 w-3.5" aria-hidden />}
   420→                Set mode to {nextMode}
   421→              </Button>
   422→              {note ? (
   423→                <p className="flex items-center gap-1.5 text-xs text-ok" role="status">
   424→                  <CircleCheck className="h-3.5 w-3.5" aria-hidden /> {note}
   425→                </p>
   426→              ) : null}
   427→              {error ? (
   428→                <Alert variant="destructive" role="alert">
   429→                  <OctagonAlert className="h-4 w-4" aria-hidden />
   430→                  <AlertTitle>Mode change rejected</AlertTitle>
   431→                  <AlertDescription>{error}</AlertDescription>
   432→                </Alert>
   433→              ) : null}
   434→            </>
   435→          ) : (
   436→            <Alert className="border-hold/40 bg-hold/10 text-foreground" role="note">
   437→              <TriangleAlert className="h-4 w-4 text-hold" aria-hidden />
   438→              <AlertTitle>Read-only</AlertTitle>
   439→              <AlertDescription>Only an administrator (and the compliance lead) can switch modes. Ask one, or read the runbook.</AlertDescription>
   440→            </Alert>
   441→          )}
   442→          <Alert className="border-sev-critical/30 bg-sev-critical/5 text-foreground" role="note">
   443→            <OctagonAlert className="h-4 w-4 text-sev-critical" aria-hidden />
   444→            <AlertTitle>Caution — safe halt (M5)</AlertTitle>
   445→            <AlertDescription>M5 blocks runs and decisions until an administrator clears it. Use it when the chain fails or data integrity is in doubt.</AlertDescription>
   446→          </Alert>
   447→          <p className="text-[11px] text-muted-foreground">{iso(String(MODES.length))} modes · current {current}</p>
   448→        </CardContent>
   449→      </Card>
   450→    </div>
   451→  );
   452→}

   532→
   533→function EvidenceTab() {
   534→  const { iso } = useI18n();
   535→  const [kind, setKind] = useState<"run" | "exception">("run");
   536→  const [exceptionKey, setExceptionKey] = useState("");
   537→  const [error, setError] = useState<string | null>(null);
   538→  const [pack, setPack] = useState<EvidencePack | null>(null);
   539→  const [busy, setBusy] = useState(false);
   540→
   541→  const runsQuery = useQuery<{ items: RunItem[] }>({
   542→    queryKey: ["runs"],
   543→    queryFn: () => api<{ items: RunItem[] }>(v1("/runs")),
   544→  });
   545→
   546→  const latestRunId = runsQuery.data?.items[0]?.id ?? null;
   547→  const scope = kind === "run" ? (latestRunId ? `run:${latestRunId}` : null) : exceptionKey.trim() ? `exception:${exceptionKey.trim()}` : null;
   548→
   549→  const onBuild = async (): Promise<void> => {
   550→    if (!scope) return;
   551→    setBusy(true);
   552→    setError(null);
   553→    try {
   554→      const result = await api<EvidencePack>(v1(`/evidence-pack?scope=${encodeURIComponent(scope)}`));
   555→      setPack(result);
   556→      const blob = new Blob([JSON.stringify(result, null, 2)], { type: "application/json" });
   557→      const url = URL.createObjectURL(blob);
   558→      const a = document.createElement("a");
   559→      a.href = url;
   560→      a.download = "evidence-pack.json";
   561→      document.body.appendChild(a);
   562→      a.click();
   563→      a.remove();
   564→      URL.revokeObjectURL(url);
   565→    } catch (e) {
   566→      setPack(null);
   567→      setError(e instanceof ApiError ? e.message : (e as Error).message);
   568→    } finally {
   569→      setBusy(false);
   570→    }
   571→  };
   572→
   573→  const fileEntries = pack ? Object.entries(pack.manifest.files) : [];
   574→
   575→  return (
   576→    <Card>
   577→      <CardHeader className="pb-3">
   578→        <CardTitle className="text-sm">Evidence pack</CardTitle>
   579→        <CardDescription>
   580→          Findings, rule pack, interpretations, audit extract and chain verification for one scope — exported with a signed manifest.
   581→        </CardDescription>
   582→      </CardHeader>
   583→      <CardContent className="space-y-4">
   584→        <RadioGroup value={kind} onValueChange={(v) => setKind(v as "run" | "exception")} className="grid gap-2 sm:grid-cols-2">
   585→          <Label
   586→            htmlFor="scope-run"
   587→            className={cn("flex cursor-pointer items-start gap-2.5 rounded-lg border p-3", kind === "run" ? "border-primary/50 bg-secondary" : "border-border")}
   588→          >
   589→            <RadioGroupItem id="scope-run" value="run" className="mt-0.5" />
   590→            <span className="space-y-0.5">
   591→              <span className="block text-sm font-medium">Latest run</span>
   592→              <span className="block font-mono text-xs text-muted-foreground">
   593→                run:{latestRunId ? trunc(latestRunId, 24) : "— no run yet, start one first"}
   594→              </span>
   595→            </span>
   596→          </Label>
   597→          <Label
   598→            htmlFor="scope-exception"
   599→            className={cn("flex cursor-pointer items-start gap-2.5 rounded-lg border p-3", kind === "exception" ? "border-primary/50 bg-secondary" : "border-border")}
   600→          >
   601→            <RadioGroupItem id="scope-exception" value="exception" className="mt-0.5" />
   602→            <span className="space-y-0.5">
   603→              <span className="block text-sm font-medium">Paste an exception key</span>
   604→              <span className="block font-mono text-xs text-muted-foreground">exception:F-XXXXXXXX…</span>
   605→            </span>
   606→          </Label>
   607→        </RadioGroup>
   608→
   609→        {kind === "exception" ? (
   610→          <div className="space-y-1.5">
   611→            <Label htmlFor="exception-key" className="text-xs">Exception key</Label>
   612→            <Input
   613→              id="exception-key"
   614→              value={exceptionKey}
   615→              onChange={(e) => setExceptionKey(e.target.value)}
   616→              placeholder="F-3KQ9M2P7XW1A"
   617→              className="max-w-sm font-mono text-xs"
   618→              aria-label="Exception key for the evidence pack scope"
   619→            />
   620→          </div>
   621→        ) : null}
   622→
   623→        <div className="flex flex-wrap items-center gap-2">
   624→          <Button size="sm" onClick={onBuild} disabled={busy || !scope}>
   625→            {busy ? <Loader2 className="me-1.5 h-3.5 w-3.5 animate-spin" aria-hidden /> : <Download className="me-1.5 h-3.5 w-3.5" aria-hidden />}
   626→            Build &amp; download evidence-pack.json
   627→          </Button>
   628→          {busy ? <span className="text-xs text-muted-foreground">Building the pack…</span> : null}
   629→        </div>
   630→
   631→        {error ? (
   632→          <Alert variant="destructive" role="alert">
   633→            <OctagonAlert className="h-4 w-4" aria-hidden />
   634→            <AlertTitle>Could not build the pack</AlertTitle>
   635→            <AlertDescription>{error}</AlertDescription>
   636→          </Alert>
   637→        ) : null}
   638→
   639→        {pack ? (
   640→          <div className="space-y-2 rounded-lg border bg-secondary/40 p-3" role="status">
   641→            <p className="flex items-center gap-1.5 text-xs">
   642→              <PackageCheck className="h-3.5 w-3.5 text-ok" aria-hidden />
   643→              Downloaded · scope <span className="font-mono">{pack.manifest.scope}</span> · signature{" "}
   644→              <span className="font-mono">{pack.manifest.signature.slice(0, 16)}</span>
   645→            </p>
   646→            <div className="overflow-hidden rounded-lg border bg-card">
   647→              <Table>
   648→                <TableHeader>
   649→                  <TableRow>
   650→                    <TableHead>File</TableHead>
   651→                    <TableHead>SHA-256</TableHead>
   652→                  </TableRow>
   653→                </TableHeader>
   654→                <TableBody>
   655→                  {fileEntries.map(([name, sha]) => (
   656→                    <TableRow key={name}>
   657→                      <TableCell className="font-mono text-xs">{name}</TableCell>
   658→                      <TableCell className="font-mono text-xs text-muted-foreground" title={sha}>{sha.slice(0, 16)}</TableCell>
   659→                    </TableRow>
   660→                  ))}
   661→                </TableBody>
   662→              </Table>
   663→            </div>
   664→          </div>
   665→        ) : null}
   666→
   667→        <p className="rounded-md bg-secondary px-3 py-2 text-xs text-muted-foreground">
   668→          Evidence packs export with a signed manifest; <span className="font-mono">mizan audit verify</span> checks them offline.
   669→        </p>
   670→      </CardContent>
   671→    </Card>
   672→  );
   673→}
