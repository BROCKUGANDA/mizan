   137→/** Run status chip (short form used inside the census banner). */
   138→function RunStateBadge({ status }: { status: string }) {
   139→  const style =
   140→    status === "COMPLETED"
   141→      ? "border-ok/40 bg-ok/10 text-foreground"
   142→      : status.endsWith("_WITH_HOLDS")
   143→        ? "border-hold/40 bg-hold/10 text-hold"
   144→        : status.endsWith("_WITH_DQ")
   145→          ? "border-sev-medium/40 bg-sev-medium/10 text-foreground"
   146→          : status === "FAILED" || status === "CANCELLED"
   147→            ? "border-destructive/40 bg-destructive/10 text-destructive"
   148→            : "border-border bg-secondary text-foreground";
   149→  return (
   150→    <span className={cn("inline-flex items-center gap-1 rounded-full border px-2 py-0.5 text-xs font-medium", style)} role="status">
   151→      {status}
   152→    </span>
   153→  );
   154→}

   156→/** A number rendered as a small font-mono badge (digits through iso()). */
   157→function Num({ value, className }: { value: string | number; className?: string }) {

   161→      {iso(String(value))}
   162→    </span>
   163→  );
   164→}
