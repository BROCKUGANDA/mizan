   182→/** Count-up numeral. variant "chip" mirrors Num, "plain" is a bare figure. */
   183→function AnimatedNum({
   184→  value,
   185→  variant = "chip",
   186→  durationMs,
   187→  className,
   188→}: {
   189→  value: string | number;
   190→  variant?: "chip" | "plain";
   191→  durationMs?: number;
   192→  className?: string;
   193→}) {
   194→  const { iso } = useI18n();
   195→  const numeric = typeof value === "number" ? value : Number(String(value).replace(/[^\d.-]/g, "")) || 0;
   196→  const fmt = useCallback((n: number) => iso(String(Math.round(n))), [iso]);
   197→  const ref = useCountUpText<HTMLSpanElement>(numeric, fmt, { durationMs: durationMs ?? 900 });
   198→  if (variant === "chip") {
   199→    return (
   200→      <span ref={ref} className={cn("inline-flex min-w-8 items-center justify-center rounded border border-border bg-secondary px-1.5 py-0.5 font-mono text-xs tabular-nums", className)} />
   201→    );
   202→  }
   203→  return <span ref={ref} className={cn("font-mono tabular-nums", className)} />;
   204→}

   206→/** Gold→brand progress ring with a count-up score; one eased rAF drives both. */
   207→function ReadinessRing({ value }: { value: number }) {
   208→  const { iso } = useI18n();
   209→  const clamped = Math.max(0, Math.min(100, value));
   210→  const r = 52;
   211→  const c = 2 * Math.PI * r;
   212→  const circleRef = useRef<SVGCircleElement | null>(null);
   213→  const fmt = useCallback((n: number) => iso(String(Math.round(n))), [iso]);
   214→  const numRef = useCountUpText<HTMLSpanElement>(clamped, fmt, {
   215→    durationMs: 1100,
   216→    onUpdate: (v) => {
   217→      const pct = Math.max(0, Math.min(100, v)) / 100;
   218→      circleRef.current?.setAttribute("stroke-dashoffset", String(c * (1 - pct)));
   219→    },
   220→  });
   221→  return (
   222→    <div className="relative h-28 w-28 shrink-0" role="img" aria-label={`${value} / 100`}>
   223→      <svg viewBox="0 0 120 120" className="h-full w-full -rotate-90" aria-hidden>
   224→        <circle cx="60" cy="60" r={r} fill="none" stroke="var(--border)" strokeWidth="8" />
   225→        <defs>
   226→          <linearGradient id="mizan-rr-grad" x1="0" y1="0" x2="1" y2="1">
   227→            <stop offset="0%" stopColor="var(--gold)" />
   228→            <stop offset="100%" stopColor="var(--brand)" />
   229→          </linearGradient>
   230→        </defs>
   231→        <circle
   232→          ref={circleRef}
   233→          cx="60"
   234→          cy="60"
   235→          r={r}
   236→          fill="none"
   237→          stroke="url(#mizan-rr-grad)"
   238→          strokeWidth="8"
   239→          strokeLinecap="round"
   240→          strokeDasharray={c}
   241→          strokeDashoffset={c}
   242→        />
   243→      </svg>
   244→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   245→        <span ref={numRef} className="font-mono text-2xl font-bold tabular-nums" />
   246→        <span className="text-[10px] text-muted-foreground">{iso("100")}</span>
   247→      </div>
   248→    </div>
   249→  );
   250→}

   252→/** Staggered entrance wrapper; reduced-motion users get a plain div. */
   253→function Reveal({ index, className, children }: { index: number; className?: string; children: React.ReactNode }) {
   254→  const reduce = useReducedMotion();
   255→  if (reduce) return <div className={className}>{children}</div>;
   256→  return (
   257→    <motion.div
   258→      className={className}
   259→      initial={{ opacity: 0, y: 12 }}
   260→      animate={{ opacity: 1, y: 0 }}
   261→      transition={{ duration: 0.32, delay: Math.min(index * 0.06, 0.42), ease: "easeOut" }}
   262→    >
   263→      {children}
   264→    </motion.div>
   265→  );
   266→}

   272→/** Time-aware greeting + Gregorian and Umm al-Qura Hijri dates (luxury, zero deps). */
   273→function GreetingHeader({ live }: { live: boolean }) {
   274→  const { locale } = useI18n();
   275→  const actor = useAppStore((s) => s.actor);
   276→  // Shares the shell's ["system-state", actor] cache — no extra polling.
   277→  const { data: system } = useQuery<SystemStateLite>({
   278→    queryKey: ["system-state", actor],
   279→    queryFn: () => api<SystemStateLite>(v1("/system/state")),
   280→    staleTime: 10_000,
   281→    refetchInterval: false,
   282→  });
   283→  const name = system?.actors.find((a) => a.id === actor)?.name ?? actor;
   284→  const now = new Date();
   285→  const h = now.getHours();
   286→  const greeting =
   287→    locale === "ar"
   288→      ? h < 12
   289→        ? "صباح الخير"
   290→        : "مساء الخير"
   291→      : h < 12
   292→        ? "Good morning"
   293→        : h < 17
   294→          ? "Good afternoon"
   295→          : "Good evening";
   296→  const gregorian = new Intl.DateTimeFormat(locale === "ar" ? "ar-AE" : "en-AE", {
   297→    weekday: "long",
   298→    day: "numeric",
   299→    month: "long",
   300→    year: "numeric",
   301→  }).format(now);
   302→  let hijri = "";
   303→  try {
   304→    hijri = new Intl.DateTimeFormat(locale === "ar" ? "ar-AE-u-ca-islamic-umalqura" : "en-u-ca-islamic-umalqura", {
   305→      day: "numeric",
   306→      month: "long",
   307→      year: "numeric",
   308→    }).format(now);
   309→  } catch {
   310→    hijri = ""; // ICU without islamic calendar — Gregorian line still shows
   311→  }
   312→  return (
   313→    <div>
   314→      <div className="flex flex-wrap items-end justify-between gap-4">
   315→        <div>
   316→          <h1 className="text-xl font-semibold tracking-tight">
   317→            {greeting}, <span className="text-brand">{name}</span>
   318→          </h1>
   319→          <p className="mt-1 text-xs text-muted-foreground">
   320→            {gregorian}
   321→            {hijri ? (
   322→              <>
   323→                <span aria-hidden> · </span>
   324→                {hijri}
   325→              </>
   326→            ) : null}
   327→          </p>
   328→        </div>
   329→        {live ? (
   330→          <span
   331→            className="inline-flex items-center gap-2 rounded-full border border-ok/40 bg-ok/10 px-3 py-1 text-xs text-foreground"
   332→            role="status"
   333→          >
   334→            <span className="pulse-dot h-2 w-2 rounded-full bg-ok" aria-hidden />
   335→            {locale === "ar" ? "التعداد محدَّث" : "Census current"}
   336→          </span>
   337→        ) : null}
   338→      </div>
   339→      <div className="gold-rule mt-4" aria-hidden />
   340→    </div>
   341→  );
   342→}
