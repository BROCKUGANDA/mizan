   206→   206→/** Gold→brand progress ring with a count-up score; one eased rAF drives both. */
   207→   207→function ReadinessRing({ value }: { value: number }) {
   208→   208→  const { iso } = useI18n();
   209→   209→  const clamped = Math.max(0, Math.min(100, value));
   210→   210→  const r = 52;
   211→   211→  const c = 2 * Math.PI * r;
   212→   212→  const circleRef = useRef<SVGCircleElement | null>(null);
   213→   213→  const fmt = useCallback((n: number) => iso(String(Math.round(n))), [iso]);
   214→   214→  const numRef = useCountUpText<HTMLSpanElement>(clamped, fmt, {
   215→   215→    durationMs: 1100,
   216→   216→    onUpdate: (v) => {
   217→   217→      const pct = Math.max(0, Math.min(100, v)) / 100;
   218→   218→      circleRef.current?.setAttribute("stroke-dashoffset", String(c * (1 - pct)));
   219→   219→    },
   220→   220→  });
   221→   221→  return (
   222→   222→    <div className="relative h-28 w-28 shrink-0" role="img" aria-label={`${value} / 100`}>
   223→   223→      <svg viewBox="0 0 120 120" className="h-full w-full -rotate-90" aria-hidden>
   224→   224→        <circle cx="60" cy="60" r={r} fill="none" stroke="var(--border)" strokeWidth="8" />
   225→   225→        <defs>
   226→   226→          <linearGradient id="mizan-rr-grad" x1="0" y1="0" x2="1" y2="1">
   227→   227→            <stop offset="0%" stopColor="var(--gold)" />
   228→   228→            <stop offset="100%" stopColor="var(--brand)" />
   229→   229→          </linearGradient>
   230→   230→        </defs>
   231→   231→        <circle
   232→   232→          ref={circleRef}
   233→   233→          cx="60"
   234→   234→          cy="60"
   235→   235→          r={r}
   236→   236→          fill="none"
   237→   237→          stroke="url(#mizan-rr-grad)"
   238→   238→          strokeWidth="8"
   239→   239→          strokeLinecap="round"
   240→   240→          strokeDasharray={c}
   241→   241→          strokeDashoffset={c}
   242→   242→        />
   243→   243→      </svg>
   244→   244→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   245→   245→        <span ref={numRef} className="font-mono text-2xl font-bold tabular-nums" />
   246→   246→        <span className="text-[10px] text-muted-foreground">{iso("100")}</span>
   247→   247→      </div>
   248→   248→    </div>
   249→   249→  );
   250→   250→}

   272→   272→/** Time-aware greeting + Gregorian and Umm al-Qura Hijri dates (luxury, zero deps). */
   273→   273→function GreetingHeader({ live }: { live: boolean }) {
   274→   274→  const { locale } = useI18n();
   275→   275→  const actor = useAppStore((s) => s.actor);
   276→   276→  // Shares the shell's ["system-state", actor] cache — no extra polling.
   277→   277→  const { data: system } = useQuery<SystemStateLite>({
   278→   278→    queryKey: ["system-state", actor],
   279→   279→    queryFn: () => api<SystemStateLite>(v1("/system/state")),
   280→   280→    staleTime: 10_000,
   281→   281→    refetchInterval: false,
   282→   282→  });
   283→   283→  const name = system?.actors.find((a) => a.id === actor)?.name ?? actor;
   284→   284→  const now = new Date();
   285→   285→  const h = now.getHours();
   286→   286→  const greeting =
   287→   287→    locale === "ar"
   288→   288→      ? h < 12
   289→   289→        ? "صباح الخير"
   290→   290→        : "مساء الخير"
   291→   291→      : h < 12
   292→   292→        ? "Good morning"
   293→   293→        : h < 17
   294→   294→          ? "Good afternoon"
   295→   295→          : "Good evening";
   296→   296→  const gregorian = new Intl.DateTimeFormat(locale === "ar" ? "ar-AE" : "en-AE", {
   297→   297→    weekday: "long",
   298→   298→    day: "numeric",
   299→   299→    month: "long",
   300→   300→    year: "numeric",
   301→   301→  }).format(now);
   302→   302→  let hijri = "";
   303→   303→  try {
   304→   304→    hijri = new Intl.DateTimeFormat(locale === "ar" ? "ar-AE-u-ca-islamic-umalqura" : "en-u-ca-islamic-umalqura", {
   305→   305→      day: "numeric",
   306→   306→      month: "long",
   307→   307→      year: "numeric",
   308→   308→    }).format(now);
   309→   309→  } catch {
   310→   310→    hijri = ""; // ICU without islamic calendar — Gregorian line still shows
   311→   311→  }
   312→   312→  return (
   313→   313→    <div>
   314→   314→      <div className="flex flex-wrap items-end justify-between gap-4">
   315→   315→        <div>
   316→   316→          <h1 className="text-xl font-semibold tracking-tight">
   317→   317→            {greeting}, <span className="text-brand">{name}</span>
   318→   318→          </h1>
   319→   319→          <p className="mt-1 text-xs text-muted-foreground">
   320→   320→            {gregorian}
   321→   321→            {hijri ? (
   322→   322→              <>
   323→   323→                <span aria-hidden> · </span>
   324→   324→                {hijri}
   325→   325→              </>
   326→   326→            ) : null}
   327→   327→          </p>
   328→   328→        </div>
   329→   329→        {live ? (
   330→   330→          <span
   331→   331→            className="inline-flex items-center gap-2 rounded-full border border-ok/40 bg-ok/10 px-3 py-1 text-xs text-foreground"
   332→   332→            role="status"
   333→   333→          >
   334→   334→            <span className="pulse-dot h-2 w-2 rounded-full bg-ok" aria-hidden />
   335→   335→            {locale === "ar" ? "التعداد محدَّث" : "Census current"}
   336→   336→          </span>
   337→   337→        ) : null}
   338→   338→      </div>
