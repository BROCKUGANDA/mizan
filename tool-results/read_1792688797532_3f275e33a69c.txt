   131→function runStatusStyle(status: string): string {
   132→  if (status === "COMPLETED") return "border-ok/40 bg-ok/10 text-foreground";
   133→  if (status.endsWith("_WITH_HOLDS")) return "border-hold/40 bg-hold/10 text-hold";
   134→  if (status.endsWith("_WITH_DQ")) return "border-sev-medium/40 bg-sev-medium/10 text-foreground";
   135→  if (status === "FAILED" || status === "CANCELLED") return "border-destructive/40 bg-destructive/10 text-destructive";
   136→  return "border-border bg-secondary text-foreground"; // QUEUED / RUNNING / FINALIZING
   137→}

   139→/** Decode base64 → bytes and trigger a client-side download. */
   140→function downloadB64(b64: string, filename: string): void {
   141→  const bin = atob(b64);
   142→  const bytes = new Uint8Array(bin.length);
   143→  for (let i = 0; i < bin.length; i++) bytes[i] = bin.charCodeAt(i);
   144→  const url = URL.createObjectURL(new Blob([bytes], { type: "text/csv;charset=utf-8" }));
   145→  const a = document.createElement("a");
   146→  a.href = url;
   147→  a.download = filename;
   148→  document.body.appendChild(a);
   149→  a.click();
   150→  a.remove();
   151→  URL.revokeObjectURL(url);
   152→}
